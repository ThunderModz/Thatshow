local function make(class, props, parent)
    local obj = Instance.new(class)
    for k, v in pairs(props) do
        obj[k] = v
    end
    obj.Parent = parent
    return obj
end

local layout = Instance.new("UIListLayout")
layout.Padding = UDim.new(0, 4)
layout.Parent = list

local counts = {}
local rows = {}

local function updateRow(itemName)
    local row = rows[itemName]

    if not row then
        row = make("TextLabel", {
            Size = UDim2.new(1, -8, 0, 27),
            BackgroundColor3 = Color3.fromRGB(31, 43, 65),
            TextColor3 = Color3.new(1, 1, 1),
            Font = Enum.Font.GothamBold,
            TextSize = 11
        }, list)

        Instance.new("UICorner", row).CornerRadius = UDim.new(0, 5)
        rows[itemName] = row
    end

    row.Text = "  " .. itemName .. "                         x" .. counts[itemName]
    list.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 6)
end

dupeButton.MouseButton1Click:Connect(function()
    local itemName = nameBox.Text:match("^%s*(.-)%s*$")
    local startingCount = tonumber(countBox.Text)

    if itemName == "" then
        status.Text = "Enter an item name."
        return
    end

    if not startingCount or startingCount < 1
        or startingCount % 1 ~= 0 then
        status.Text = "Enter a valid starting count."
        return
    end

    if counts[itemName] == nil then
        counts[itemName] = startingCount
    end

    counts[itemName] += 1
    updateRow(itemName)

    status.Text = itemName .. " count: x" .. counts[itemName]
end)
